# The pinned runtime set for a Mantle deployment: the resolved closure of `pyproject.toml`'s
# `[service]` extra.
#
# `[service]` states the contract in ranges; this states one reproducible resolution of it, so a
# deployment can install a known set rather than whatever the index resolves that day. Two lists
# that name different packages would make "what the service needs" a question with two answers, so
# keep them in step: a package added to `[service]` is added here, and nothing is added here that
# `[service]` does not name.
#
# Test tooling is not a runtime requirement and is declared in `[dev]` instead. CI installs
# `-e '.[dev]'` alongside this file.
#
# `agience-prism` is the one `[service]` requirement absent here on purpose: CI installs it from a
# checkout so the suite runs against this organization's prism revision.

# ── base install ─────────────────────────────────────────────────────────────
# The store encrypts at rest, so this is a hard dependency, not an extra.
cryptography==46.0.7

# ── the web service ──────────────────────────────────────────────────────────
# `fastapi` accepts a RANGE of starlette versions, so leaving it open meant this file pinned the
# framework's dependency and not the framework — a build months apart from another could take a
# different FastAPI against the same pinned starlette, which is the one outcome the header rules
# out. Pinned to what the resolver picks today, against these pins.
fastapi==0.141.1
starlette==0.49.3
uvicorn[standard]==0.37.0
python-multipart==0.0.26
python-dotenv==1.2.2
itsdangerous==2.2.0

# ── validation ───────────────────────────────────────────────────────────────
pydantic==2.12.0
email-validator==2.2.0

# ── auth ─────────────────────────────────────────────────────────────────────
# The `[cryptography]` extra is load-bearing: jose needs a backend to verify RS256 at all.
python-jose[cryptography]==3.5.0
authlib==1.6.9
bcrypt==5.0.0

# ── seed provisioning and addon manifests (`[service]`) ──────────────────────
# `services/seed_provisioning/loader.py` and `system/manage_addons.py` read YAML.
pyyaml==6.0.3

# ── the vector arm (`[semantic]`) ────────────────────────────────────────────
# The arm's whole weight. It loads an AnchorSet from JSON and routes against it.
numpy==2.3.3

# ── the durable content tier (`[s3]`) ────────────────────────────────────────
# `botocore` is boto3's own dependency, pinned in lockstep because mantle imports it directly
# (`from botocore.exceptions import ClientError`) rather than only through boto3.
boto3==1.40.50
botocore==1.40.50

# ── outbound HTTP ────────────────────────────────────────────────────────────
httpx==0.28.1
